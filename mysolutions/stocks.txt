class Solution {
public:
    int maxProfit(vector<int>& arr) {
        int n=arr.size();
        int minPrice = 1e9;
        int ans=0;
        for(int i=0;i<n;i++){
            ans=max(ans,arr[i]-minPrice);
            minPrice = min(minPrice,arr[i]);
        }
        return ans;
    }
};

class Solution {
public:
    int maxProfit(vector<int>& arr) {
        int mintillnow = 1e9;
        int ans=0;
        for(auto x:arr){
            if(x > mintillnow){
                ans += x-mintillnow;
                mintillnow=x;
            }
            else{
                mintillnow = x;
            }
        }
        return ans;
    }
};

Q3.Version 1
class Solution {
public:

    int dp[100005][3][2];
    int n;
    int rec(int i,int remk,int state,vector<int>& arr){
        if(i==n) return 0;
        if(dp[i][remk][state]!=-1) return dp[i][remk][state];
        int ans = rec(i+1,remk,state,arr);
        if(state==0 && remk) ans=max(ans,-arr[i]+rec(i+1,remk-1,1,arr));
        if(state==1) ans=max(ans,arr[i]+rec(i+1,remk,0,arr));
        return dp[i][remk][state]=ans;
    }


    int maxProfit(vector<int>& arr) {
        // atmost two transactions can be done 
        n=arr.size();
        memset(dp,-1,sizeof(dp));
        return rec(0,2,0,arr);
    }
};

Version 2 
class Solution {
public:
    int maxProfit(vector<int>& arr) {
        int n = arr.size();
        int dp[100005][3][2];

        for (int remk = 0; remk <= 2; remk++) {
            for (int state = 0; state < 2; state++) {
                dp[n][remk][state] = 0;
            }
        }

        for (int i = n - 1; i >= 0; i--) {
            for (int remk = 0; remk <= 2; remk++) {
                for (int state = 0; state < 2; state++) {
                    int ans = dp[i + 1][remk][state];
                    if (state == 0 && remk)
                        ans = max(ans, -arr[i] + dp[i + 1][remk - 1][1]);
                    if (state == 1)
                        ans = max(ans, arr[i] + dp[i + 1][remk][0]);
                    dp[i][remk][state] = ans;
                }
            }
        }

        return max(dp[0][1][0],dp[0][2][0]);
    }
};

class Solution {

        int after[3][2];

        for (int remk = 0; remk <= 2; remk++) {
            for (int state = 0; state < 2; state++) {
                after[remk][state] = 0;
            }
        }

        for (int i = n - 1; i >= 0; i--) {
            for (int remk = 0; remk <= 2; remk++) {
                for (int state = 0; state < 2; state++) {
                    int ans = after[remk][state];
                    if (state == 0 && remk)
                        ans = max(ans, -arr[i] + after[remk - 1][1]);
                    if (state == 1)
                        ans = max(ans, arr[i] + after[remk][0]);

                }
            }
            swap(after,before);
        }

        return max(after[1][0],after[2][0]);
    }

        long long after[503][3];

        for (int remk = 0; remk <= k; remk++) {
            after[remk][0] = 0;
            after[remk][1] = 0;
            after[remk][2] = -1e11; // nuksan 
        }

        for (int i = n - 1; i >= 0; i--) {
            for (int remk = 0; remk <= k; remk++) {
                for (int state = 0; state <= 2; state++) {
                    long long ans = after[remk][state];
                    if (state == 0 && remk)
                        ans = max({ans, -arr[i] + after[remk - 1][1],+arr[i] + after[remk-1][2]});
                    if (state == 1)    ans = max(ans, arr[i] + after[remk][0]);
                    if (state == 2)    ans = max(ans, -arr[i] + after[remk][0]);
                    before[remk][state] = ans;
                }
            }
            swap(after,before);
        }


        long long ans=0;
        for(int remk=0;remk<=k;remk++){
            ans=max(ans,after[remk][0]);
        }
        return ans;
    }

Q6. Cooldown 
class Solution {
public:
    int maxProfit(vector<int>& arr) {
        int n = arr.size();
        int before[2];
        int after[2];
        int nextafter[2];

        for (int state = 0; state < 2; state++) {
            nextafter[state] = 0;
            after[state] = 0;
        }

        for (int i = n - 1; i >= 0; i--) {
            for (int state = 0; state < 2; state++) {
                int ans = after[state];
                if (state == 0 ) ans = max(ans, -arr[i] + after[1]);
                if (state == 1) ans = max(ans, arr[i] + nextafter[0]);
                before[state] = ans;
            }
            swap(after,nextafter);
            swap(after,before);
        }

        return after[0];
    }
};


Q7. Greedy 
class Solution {
public:
    long long maxProfit(vector<int>& prices, vector<int>& startegy, int k) {
        // atmost one modification    
        int n=prices.size();
        vector<long long>simpleStrategy(n,prices[0]*startegy[0]);
        vector<long long>pre(n,prices[0]);
        for(int i=1;i<n;i++){
            simpleStrategy[i]  = simpleStrategy[i-1] + prices[i]*startegy[i];
            pre[i]  = pre[i-1] + prices[i];
        }
        long long normalStrategy = simpleStrategy[n-1];
        long long ans=(pre[k-1]-pre[k/2-1])-(simpleStrategy[k-1]);
        for(int i=1;k-1+i<n;i++){
            ans=max(ans,(pre[k-1+i]-pre[k/2-1+i])-(simpleStrategy[k-1+i]-simpleStrategy[i-1]));
        }
        return max(normalStrategy,normalStrategy+ans);
    }
};


Q8. Transaction fee
class Solution {
public:
    int maxProfit(vector<int>& arr, int fee) {
        int n = arr.size();
        int before[2];
        int after[2];

        for (int state = 0; state < 2; state++) {
            after[state] = 0;
        }

        for (int i = n - 1; i >= 0; i--) {
            for (int state = 0; state < 2; state++) {
                int ans = after[state];
                if (state == 0 ) ans = max(ans, -arr[i]-fee + after[1]);
                if (state == 1) ans = max(ans, arr[i] + after[0]);
                before[state] = ans;
            }
            swap(after,before);
        }

        return after[0];
    }
};

Q9. cooldown 
class Solution {
public:
    int dp[1505];
   int n;
   int cooldown;
   vector<int> costs,prices;
    int rec(int i){
        if(i>=n) return 0;
        if(dp[i]!=-1) return dp[i];
        int ans=rec(i+1);
        for(int j=i+1;j<n;j++){
            ans=max(ans,prices[j] - prices[i] - costs[j - i]+ rec(j + cooldown + 1));
        }
        return dp[i]=ans;
    }



    int maxProfit(vector<int>& prices_, int cooldown_, vector<int>& costs_) {
        costs=costs_;
        prices=prices_;
        cooldown=cooldown_;
        n=prices.size();
        memset(dp, -1, sizeof(dp));
        return rec(0);
    }
};

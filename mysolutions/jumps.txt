class Solution {
public:
    bool canJump(vector<int>& arr) {
        int n=arr.size();
        int i=0;
        int bandwidth = 0;
        while(i<n-1){
            bandwidth=max(bandwidth,arr[i]);
            if(!bandwidth) return 0;
            bandwidth--;
            i++;
        }
        return true;
    }
};


class Solution {
public:
    int jump(vector<int>& arr) {
        int n=arr.size();
        // jitni baar paisa khatm hoga utne paisa chahiye 
        if(n==1) return 0;
        int i=0;
        int ans=0;
        int currpower=0;
        int maxpowergathered=0;
        while(i<n-1){ // assumes ki i par to ham aa skte hai
            maxpowergathered=max(maxpowergathered-1,arr[i]);
            if(currpower==0){
                currpower = maxpowergathered;
                ans++;
            }
            currpower--;
            i++;
        }
        return ans;
    }
};

class Solution {
public:
    vector<int>vis;
    int n;
    bool ans=0;
    void dfs(int i,vector<int>& arr){
        if(i<0 || i>=n) return;
        if(vis[i]) return;
        vis[i]=1;
        if(arr[i]==0){
            ans=1;
            return;
        }
        dfs(i-arr[i],arr);
        dfs(i+arr[i],arr);
    }

    bool canReach(vector<int>& arr, int start) {
        n=arr.size();
        vis.assign(n,0);
        dfs(start,arr);
        return ans;
    }
};


class Solution {
public:
    map<int,vector<int>>mp; // val, {idx,1}
    map<pair<int,int>,int>dist;
    int n;
    void bfs01(vector<int>& arr){
        deque<pair<int,int>>dq; // 0->idx, 1->value
        dq.push_back({0,0});

        dist[{0,0}]=0;

        while(!dq.empty()){
            auto [i,tag] = dq.front(); dq.pop_front();
            
            if(tag){
                for(auto idx:mp[i]){
                    if(!dist.count({idx,0}) ||  dist[{idx,0}]>dist[{i,tag}]){
                        dist[{idx,0}] = dist[{i,tag}];
                        dq.push_front({idx,0});
                    }
                }
            }


            else{

                if(i+1<n &&(!dist.count({i+1,0}) ||  dist[{i+1,0}]>dist[{i,0}]+1)){
                    dist[{i+1,0}] = dist[{i,0}]+1;
                    dq.push_back({i+1,0});
                }
                if(i-1>=0 &&(!dist.count({i-1,0}) ||  dist[{i-1,0}]>dist[{i,0}]+1)){
                    dist[{i-1,0}] = dist[{i,0}]+1;
                    dq.push_back({i-1,0});
                }
                int val = arr[i];
                if(!dist.count({val,1}) ||  dist[{val,1}]>dist[{i,0}]+1){
                    dist[{val,1}]=dist[{i,0}]+1;
                    dq.push_back({val,1});
                }

            }

        }


    }

    int minJumps(vector<int>& arr) {
        /// number of unique values 
        n=arr.size();
        for(int i = 0; i < n; i++) {
            mp[arr[i]].push_back(i);
        }
        bfs01(arr);
        return dist[{n-1,0}];
    }
};

class mdq{
public:
    deque<int>dq;
    void insert(int val){
        while(!dq.empty() && dq.back() < val) dq.pop_back();
        dq.push_back(val);
    }

    int maxv(){
        return dq.front();
    }

    void remove(int val){
        if(dq.front() == val) dq.pop_front();
    }

};

class Solution {
public:
    int maxResult(vector<int>& arr, int k) {
        int n=arr.size();
        vector<int>dp(n,0);
        mdq md;
        dp[0]=arr[0];
        md.insert(dp[0]);
        for(int i=1;i<n;i++){
            dp[i]  = arr[i]  + md.maxv();
            md.insert(dp[i]);
            if(i-k>=0) md.remove(dp[i-k]);
        }
        return dp[n-1];
    }
};


class Solution {
public:
    bool canReach(string s, int minj, int maxj) {
        int n=s.size();
        queue<int>q;
        vector<bool>dp(n,0);
        dp[0]=1;
        if(minj==1) q.push(1);
        for(int i=1;i<n;i++){
            dp[i] =( s[i]=='0') &&  !q.empty();
            if(i+1-minj>=0 && dp[i+1-minj]) q.push(1);
            if(i-maxj>=0 && dp[i-maxj]) q.pop();
        }
        return dp[n-1];
    }
};




class Solution {
public:
    vector<int>ngeIdx,nseIdx;

    int n;
    long long dp[100005];
    vector<int> costs;
    long long rec(int i){
        if(i==n) return 1e11;
        if(i==n-1) return costs[i];
        if(dp[i]!=-1) return dp[i];
        return dp[i]=costs[i]+min(rec(ngeIdx[i]),rec(nseIdx[i]));
    }

    long long minCost(vector<int>&arr, vector<int>& cost) {
        costs=cost;
        n=arr.size();
        ngeIdx.assign(n,n);
        nseIdx.assign(n,n);
        stack<int>st;
        memset(dp,-1,sizeof(dp));
        for(int i=n-1;i>=0;i--){
            while(!st.empty() && arr[st.top()] < arr[i]) st.pop();
            if(!st.empty()) ngeIdx[i]=st.top();
            st.push(i);
        }
        while(!st.empty()) st.pop();
        for(int i=n-1;i>=0;i--){
            while(!st.empty() && arr[st.top()] >= arr[i]) st.pop();
            if(!st.empty()) nseIdx[i]=st.top();
            st.push(i);
        }
        return rec(0)-cost[0];
    }
};


class Solution {
public:
    vector<int> maxValue(vector<int>& arr) {
        int n=arr.size();
        vector<int>pre(n,arr[0]);
        vector<int>ans(n,-1);
        for(int i=1;i<n;i++){
            pre[i]=max(pre[i-1],arr[i]);
        }
        int i=n-1;
        while(i>=0){
            int j=i;
            int val = pre[j];
            ans[j]=val;
            int minv = arr[i];
            while(j-1>=0 && pre[j-1] > minv){
                ans[j-1]=val;
                minv=min(minv,arr[j-1]);
                j--;
            }
            i=j-1;
        }
        return ans;

    
    
    
    }

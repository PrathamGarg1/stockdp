Q2.
class Solution {
public:
    int n;
    int dp[102][102][2];
    int rec(int i,int m,int turn,vector<int>& arr){
        if(i==n) return 0;
        if(dp[i][m][turn]!=-1) return dp[i][m][turn];
        int ans =  turn ? -1e9 : 1e9;
        int profit=0;
        for(int j=i;j<i+2*m && j<n;j++){
            // take all these stones 
            profit += arr[j];
            int x = j-i+1;
            if(turn) ans=max(ans,profit + rec(j+1,max(m,x),!turn,arr));
            else ans=min(ans,rec(j+1,max(m,x),!turn,arr));
        }
        return dp[i][m][turn]=ans;
    }

    int stoneGameII(vector<int>& arr) {
        n=arr.size();
        memset(dp,-1,sizeof(dp));
        return rec(0,1,1,arr);
    }
};

class Solution {
public:
    int n;
    int dp[50005][2];
    int rec(int i,bool turn,vector<int>& arr){
        if(i==n) return 0;
        if(dp[i][turn]!=-1) return dp[i][turn];
        int profit=0;
        int ans =  turn ? -1e9 : 1e9;
        for(int j=i;j<i+3 && j<n;j++){
            profit += arr[j];
            if(turn) ans=max(ans,profit + rec(j+1,!turn,arr));
            else ans=min(ans,rec(j+1,!turn,arr));
        }
        return dp[i][turn]=ans;
    }


    string stoneGameIII(vector<int>& arr) {
        n=arr.size();
        memset(dp,-1,sizeof(dp));
        int tot=0;
        for(auto x:arr) tot +=x;
        int val = rec(0,1,arr);
        if(val > tot-val) return "Alice";
        if(val == tot-val) return "Tie";
        return "Bob";
        
    }
};

class Solution {
public:

    int dp[100005];

    int rec(int n){
        if(n==0) return 0;
        if(dp[n]!=-1) return dp[n];
        bool ans=false;
        for(int f=1;f*f<=n;f++){
            if(!rec(n-f*f)) ans=1;
        }
        return dp[n]=ans;
    }
    bool winnerSquareGame(int n) {
        memset(dp,-1,sizeof(dp));
        return rec(n);
    }
};

Q5.

class Solution {
public:
    int n;
    int dp[505][505];
    vector<int>pre;

    int rec(int i,int j,vector<int>&arr){
        if(i>=j) return 0;
        if(dp[i][j]!=-1) return dp[i][j];
        int tot = pre[j] - (i>0 ? pre[i-1] : 0);
        int ans = 0;
        for(int k=i;k<j;k++){
            int part1 = pre[k]- (i>0 ? pre[i-1] : 0);
            int part2 = tot - part1;
            if(part1 > part2){
                ans=max(ans,part2 + rec(k+1,j,arr));
            }
            else if(part1<part2){
                ans=max(ans,part1 + rec(i,k,arr));
            }
            else{
                ans=max({ans,part1 + rec(i,k,arr),part2 + rec(k+1,j,arr)});
            }
        }
        return dp[i][j] = ans;

    }

    int stoneGameV(vector<int>& arr) {
        n=arr.size();
        memset(dp,-1,sizeof(dp));
        pre.assign(n,arr[0]);
        for(int i=1;i<n;i++) pre[i]=pre[i-1]+arr[i];
        return rec(0,n-1,arr);
    }
};



Q6.

class Solution {
public:
    int stoneGameVI(vector<int>& a, vector<int>& b) {
        // a-b karo fir max nikalo 
        int n=a.size();
        vector<pair<int,int>>arr;
        for(int i=0;i<n;i++) arr.push_back({a[i],b[i]});
        sort(arr.begin(),arr.end(),[&](pair<int,int>&a,pair<int,int>&b){
            return a.first+a.second>b.first+b.second;
        });
        int ca=0;
        int cb=0;
        for(int i=0;i<n;i++){
            if(i%2==0) ca += arr[i].first;
            else cb += arr[i].second;
        }
        if(ca>cb) return 1;
        if(ca<cb) return -1;
        return 0;
    }
};


Q7.

class Solution {
public:

    long long dp[1505][1505][2];
    vector<int> arr;
    vector<long long>pre;

    long long val(int idx,int l,int r){
        return pre[r]-(l>0 ? pre[l-1] : 0)-arr[idx];
    }


    long long rec(int l,int r,bool turn){
        if(l>r) return 0;
        if(dp[l][r][turn]!=-1) return dp[l][r][turn];
        if(turn==1){
            return dp[l][r][turn] = max(val(l,l,r) + rec(l+1,r,!turn),val(r,l,r)+rec(l,r-1,!turn));
        }
        return dp[l][r][turn] = min(rec(l+1,r,!turn)-val(l,l,r),rec(l,r-1,!turn)-val(r,l,r));
    }


    int stoneGameVII(vector<int>& ar) {
        arr=ar;
        memset(dp,-1,sizeof(dp));
        int n=arr.size();
        pre.assign(n,arr[0]);
        for(int i=1;i<n;i++) pre[i]=pre[i-1]+arr[i];

        return rec(0,n-1,1);
    }
};

class Solution {
public:

    int stoneGameVIII(vector<int>& arr) {

        int n=arr.size();
        vector<int>pre;
        pre.assign(n,arr[0]);
        for(int i=1;i<n;i++){
            pre[i]=pre[i-1]+arr[i];
        }


        vector<vector<int>>dp(n+1,vector<int>(2,0));
        dp[n][1]=0;
        dp[n][0]=0;

        vector<int>maxdp(n+1,0);
        maxdp[n]=pre[n-1];

        vector<int>mindp(n+1,0);
        mindp[n]=-pre[n-1];

        for(int i=n-1;i>=1;i--){
            for(int turn=0;turn<2;turn++){
                dp[i][turn]= turn ? (maxdp[i+1]) :( mindp[i+1]);
            }
            maxdp[i] = max(maxdp[i+1],pre[i-1]+dp[i][0]);
            mindp[i] = min(mindp[i+1],-pre[i-1]+dp[i][1]);
        }


        return dp[1][1];
    }
};
